-- sa_cert_1_own_jobs (11-11-2026): on-/off-hire bunker certificate for surveyors' own jobs.
-- Achterwaarts compatibel: GBQS-certificaten (issuer 'gbqs') werken als voorheen.

-- 1. Kolommen
alter table public.hire_cert_cases
  add column if not exists surveyor_job_id uuid references public.surveyor_jobs(id) on delete set null,
  add column if not exists company_id uuid references public.surveyor_companies(id) on delete set null,
  add column if not exists issuer text not null default 'gbqs';
do $$ begin
  if not exists (select 1 from pg_constraint where conname = 'hire_cert_cases_issuer_check') then
    alter table public.hire_cert_cases add constraint hire_cert_cases_issuer_check check (issuer in ('gbqs','surveyor'));
  end if;
end $$;
create index if not exists hire_cert_cases_surveyor_job_idx on public.hire_cert_cases(surveyor_job_id);

-- 2. Dossier voor een eigen job
create or replace function public.hire_cert_ensure_case_sa(p_survey uuid)
returns uuid language plpgsql security definer set search_path to 'public' as $function$
declare s record; j surveyor_jobs; sc surveyor_clients; cid uuid; kind text; pf uuid; pt uuid; cert uuid; v_vessel uuid; v_to text; v_cc text;
begin
  select * into s from surveys where id = p_survey;
  if not found or s.surveyor_job_id is null then return null; end if;
  if s.survey_type not in ('on_hire_bunker','off_hire_bunker','on_hire_bc','off_hire_bc') then return null; end if;
  select id into cid from hire_cert_cases where survey_id = p_survey;
  if cid is not null then return cid; end if;
  select * into j from surveyor_jobs where id = s.surveyor_job_id;
  if j.id is null then return null; end if;
  select * into sc from surveyor_clients where id = j.surveyor_client_id;
  v_vessel := coalesce(j.vessel_id, (select id from vessels where imo_number = j.imo_number limit 1));
  -- e-mail schip: laatst gebruikte adres van dit bedrijf voor dit IMO
  select hc.request_to into v_to from hire_cert_cases hc join surveyor_jobs j2 on j2.id = hc.surveyor_job_id
   where hc.company_id = j.company_id and j2.imo_number = j.imo_number and hc.request_to is not null
   order by hc.requested_at desc nulls last limit 1;
  select lower(trim(e)) into v_cc from unnest(coalesce(sc.report_emails, '{}'::text[])) e where trim(e) like '%@%' limit 1;
  insert into hire_cert_cases(survey_id, order_id, vessel_id, survey_completed_at, request_to, request_cc, surveyor_job_id, company_id, issuer)
  values (p_survey, null, v_vessel, hire_cert_survey_time(p_survey), v_to, v_cc, j.id, j.company_id, 'surveyor')
  returning id into cid;
  insert into hire_cert_parties(case_id, pkey, seq, role, company, is_client)
  select cid, 'p' || row_number() over (order by sp.sort_order, sp.created_at), row_number() over (order by sp.sort_order, sp.created_at),
         case sp.role when 'owners' then 'owners' when 'charterers' then 'charterers' when 'sub_charterers' then 'sub_charterers' else 'other' end,
         sp.company_name, coalesce(lower(trim(sp.company_name)) = lower(trim(sc.name)), false)
  from survey_parties sp where sp.survey_id = p_survey and coalesce(trim(sp.company_name),'') <> '';
  if not exists (select 1 from hire_cert_parties where case_id = cid) then
    insert into hire_cert_parties(case_id, pkey, seq, role, company, is_client) values
      (cid, 'p1', 1, 'owners', coalesce(nullif(trim(s.owner),''), 'Owners'), coalesce(lower(trim(s.owner)) = lower(trim(sc.name)), false)),
      (cid, 'p2', 2, 'charterers', coalesce(nullif(trim(s.charterer),''), 'Charterers'), coalesce(lower(trim(s.charterer)) = lower(trim(sc.name)), false));
  end if;
  kind := case when s.survey_type like 'on_hire%' then 'delivery' else 'redelivery' end;
  select id into pf from hire_cert_parties where case_id = cid and role = case when kind = 'delivery' then 'owners' else 'charterers' end order by seq limit 1;
  select id into pt from hire_cert_parties where case_id = cid and role = case when kind = 'delivery' then 'charterers' else 'owners' end order by seq limit 1;
  insert into hire_certificates(case_id, letter, kind, from_party_id, to_party_id, point_code)
  values (cid, 'A', kind, pf, pt, case when kind = 'delivery' then 'APS' else 'DLOSP' end) returning id into cert;
  insert into hire_certificate_grades(certificate_id, grade, sort, measured_mt)
  select cert, m.grade, m.sort, m.measured_mt from hire_cert_measured(p_survey) m;
  return cid;
end $function$;
revoke all on function public.hire_cert_ensure_case_sa(uuid) from public, anon, authenticated;

-- 3. Bestaande functies via gecontroleerde tekstvervanging
do $$
declare d text; n text;
begin
  -- 3a. ensure_case: eigen job -> surveyortak
  d := pg_get_functiondef('public.hire_cert_ensure_case(uuid)'::regprocedure);
  n := replace(d, 'if s.order_id is null then return null; end if;  -- own jobs of surveyors: no GlobalBQS certificate case',
                  'if s.order_id is null then return hire_cert_ensure_case_sa(p_survey); end if;  -- own jobs: certificate issued by the survey company (sa_cert_1)');
  if n = d then raise exception 'anker ensure_case niet gevonden'; end if;
  execute n;
  -- 3b. calc: referentieverbruik alleen uit dossiers van dezelfde uitgever (GBQS) of hetzelfde surveybedrijf
  d := pg_get_functiondef('public.hire_cert_calc(uuid)'::regprocedure);
  n := replace(d, 'where hc.status = ''issued'' and hcc.vessel_id = cs.vessel_id and hcc.id <> cs.id and hg.grade = g.grade and hg.avg_mt_day is not null;',
                  'where hc.status = ''issued'' and hcc.vessel_id = cs.vessel_id and hcc.id <> cs.id and hg.grade = g.grade and hg.avg_mt_day is not null
         and hcc.issuer = cs.issuer and hcc.company_id is not distinct from cs.company_id;');
  if n = d then raise exception 'anker calc niet gevonden'; end if;
  execute n;
  -- 3c. open lijst ops: alleen GBQS-dossiers
  d := pg_get_functiondef('public.hire_cert_open_list()'::regprocedure);
  n := replace(d, 'where cc.status in (''open'',''requested'',''received'') order by cc.created_at',
                  'where cc.status in (''open'',''requested'',''received'') and cc.issuer = ''gbqs'' order by cc.created_at');
  if n = d then raise exception 'anker open_list niet gevonden'; end if;
  execute n;
end $$;

-- 4. case_get met uitgever en klant van de surveyor
create or replace function public.hire_cert_case_get(p_case uuid)
returns jsonb language sql stable security definer set search_path to 'public' as $function$
  select jsonb_build_object(
    'case', to_jsonb(cc) - 'evidence_path' || jsonb_build_object('has_evidence', cc.evidence_path is not null),
    'survey', jsonb_build_object('id',s.id,'gbqs_reference',s.gbqs_reference,'survey_type',s.survey_type,'vessel_name',coalesce(v.vessel_name,j.vessel_name,s.vessel_name),
        'imo',coalesce(v.imo_number,j.imo_number,s.imo_number),'place',s.place_of_survey,'surveyor',s.bunker_surveyor,'report_number',s.report_number,
        'client',coalesce(cl.company, sc.name),'client_id',cl.id,'is_on_hire', s.survey_type like 'on_hire%'),
    'issuer', case when cc.issuer = 'surveyor' then jsonb_build_object('kind','surveyor','company_id',co.id,'name',co.name,'office_email',co.office_email,
        'phone',co.phone,'website',co.website,'city',co.city,'logo_path',co.logo_path,'job_id',j.id,'job_reference',j.reference,
        'client_emails', to_jsonb(coalesce(sc.report_emails,'{}'::text[])), 'client_contact', sc.contact_name)
      else jsonb_build_object('kind','gbqs') end,
    'measured', coalesce((select jsonb_agg(jsonb_build_object('grade',m.grade,'measured',m.measured_mt) order by m.sort) from hire_cert_measured(s.id) m),'[]'),
    'parties', coalesce((select jsonb_agg(jsonb_build_object('id',p.id,'key',p.pkey,'seq',p.seq,'role',p.role,'role_label',p.role_label,
        'label',hire_cert_role_label(p.role,p.role_label),'company',p.company,'cp_date',p.cp_date,'is_client',p.is_client) order by p.seq)
        from hire_cert_parties p where p.case_id = cc.id and not p.removed),'[]'),
    'certificates', coalesce((select jsonb_agg(to_jsonb(hc) - 'calc' || jsonb_build_object(
        'from_key',(select pkey from hire_cert_parties where id = hc.from_party_id),
        'to_key',(select pkey from hire_cert_parties where id = hc.to_party_id),
        'grades',(select coalesce(jsonb_agg(to_jsonb(g) order by g.sort),'[]') from hire_certificate_grades g where g.certificate_id = hc.id),
        'check', case when hc.status = 'issued' then hc.calc else hire_cert_calc(hc.id) end) order by hc.letter)
        from hire_certificates hc where hc.case_id = cc.id and hc.status <> 'cancelled'),'[]'))
  from hire_cert_cases cc join surveys s on s.id = cc.survey_id
  left join orders o on o.id = cc.order_id left join vessels v on v.id = coalesce(cc.vessel_id, o.vessel_id)
  left join clients cl on cl.id = o.client_id
  left join surveyor_jobs j on j.id = cc.surveyor_job_id
  left join surveyor_clients sc on sc.id = j.surveyor_client_id
  left join surveyor_companies co on co.id = cc.company_id
  where cc.id = p_case $function$;

-- 5. Mailopmaak surveybedrijf
create or replace function public.sa_cert_logo_url(p_survey uuid)
returns text language sql stable security definer set search_path to 'public' as $function$
  select case when co.logo_path is not null and l.token is not null
    then coalesce((select value from system_settings where key = 'sa_report_link_base'), '') || '?logo=' || l.token end
  from surveys s join surveyor_jobs j on j.id = s.surveyor_job_id join surveyor_companies co on co.id = j.company_id
  left join lateral (select token from sa_report_links where survey_id = s.id and revoked_at is null order by created_at desc limit 1) l on true
  where s.id = p_survey $function$;

        gl text := ''; td1 text := 'padding:5px 12px 5px 0;color:#5c6b85;vertical-align:top;white-space:nowrap;';
        co surveyor_companies; v_subject text; v_title text;
begin

  for m in select * from jsonb_array_elements(d->'measured') loop
    rob := rob || gbqs_h(m->>'grade') || ' ' || to_char((m->>'measured')::numeric, 'FM999G990D000') || ' MT<br>';

  end loop;
  body :=
    '<p style="margin:0 0 14px;">Dear Captain,</p>'
    || case when p_reminder then '<p style="margin:0 0 14px;color:#b45309;font-weight:bold;">Reminder: we have not yet received the information below.</p>' else '' end
    || '<p style="margin:0 0 14px;">Following the ' || gbqs_h(lower(hire_cert_survey_label(sv->>'survey_type'))) || ' on board <strong>' || gbqs_h(sv->>'vessel_name')
    || '</strong>, we will issue the ' || case when onh then 'on-hire' else 'off-hire' end || ' bunker certificate stating the bunkers on board at the moment of ' || ev
    || '. Please reply with the figures below from the deck and engine log.</p>'
    || '<table width="100%" cellpadding="0" cellspacing="0" style="background:#f5f8fc;border:1px solid #e2e7f2;border-radius:8px;margin:0 0 20px;"><tr><td style="padding:10px 18px;"><table width="100%" cellpadding="0" cellspacing="0">'

    || gbqs_mail_row('SURVEY', gbqs_h(coalesce(sv->>'place','-')) || ' &middot; completed ' || coalesce(to_char((cs->>'survey_completed_at')::timestamptz at time zone 'UTC','DD-MM-YYYY HH24:MI') || ' UTC','-'))
    || gbqs_mail_row('ROB MEASURED', rob)
    || '</table></td></tr></table>'
    || '<table width="100%" cellpadding="0" cellspacing="0" style="margin:0 0 20px;font-size:14px;">'
    || '<tr><td colspan="2" style="padding:0 0 6px;font-weight:bold;color:#052844;">1. Moment of ' || ev || '</td></tr>'
    || '<tr><td style="' || td1 || '">' || case when onh then 'APS / DLOSP / berth' else 'DLOSP / berth' end || '</td><td style="padding:5px 0;"><span style="' || fill || '">&nbsp;</span></td></tr>'
    || '<tr><td style="' || td1 || '">Date and time (UTC)</td><td style="padding:5px 0;"><span style="' || fill || '">&nbsp;</span></td></tr>'
    || '<tr><td colspan="2" style="padding:16px 0 6px;font-weight:bold;color:#052844;">2. Consumption ' || case when onh then 'from ' || ev || ' to survey' else 'from survey to ' || ev end || ' (MT)</td></tr>'
    || gl
    || '<tr><td style="' || td1 || '">Bunkers received / discharged</td><td style="padding:5px 0;"><span style="' || fill || '">&nbsp;</span></td></tr>'
    || '<tr><td colspan="2" style="padding:16px 0 6px;font-weight:bold;color:#052844;">3. Parties</td></tr>'

-- 7. Mail bij uitgifte (GBQS ongewijzigd; surveybedrijf: klant + aparte mail aan het schip)
create or replace function public.hire_cert_mail_issued(p_case uuid)
returns jsonb language plpgsql stable security definer set search_path to 'public' as $function$
declare d jsonb := hire_cert_case_get(p_case); sv jsonb := d->'survey'; iss jsonb := d->'issuer'; c jsonb; g jsonb; rows text := ''; gr text; body text;
        fp text; tp text; to_addr text; cc jsonb; n int := 0; co surveyor_companies; v_to text; v_cc text[]; v_view text; v_logo text;
        v_ship text; v_extra jsonb := '[]'; v_kindword text;
begin
  for c in select * from jsonb_array_elements(d->'certificates') loop
    if c->>'status' <> 'issued' then continue; end if;
    n := n + 1;
    select company into fp from hire_cert_parties where id = (c->>'from_party_id')::uuid;
    select company into tp from hire_cert_parties where id = (c->>'to_party_id')::uuid;
    gr := '';
    for g in select * from jsonb_array_elements(c->'grades') loop
      gr := gr || gbqs_h(g->>'grade') || ' <strong>' || to_char((g->>'calc_mt')::numeric,'FM999G990D000') || '</strong> MT &nbsp; ';
    end loop;
    rows := rows || '<tr><td style="padding:12px 14px;border-top:1px solid #e2e7f2;vertical-align:top;">'
      || '<div style="font-family:Courier New,monospace;font-weight:bold;color:#2a8fc4;">' || gbqs_h(c->>'cert_number') || '</div>'
      || '<div style="font-weight:bold;margin:2px 0;">' || case when c->>'kind' = 'delivery' then 'On-hire' else 'Off-hire' end || ' &middot; '
      || initcap(c->>'kind') || ' ' || gbqs_h(fp) || ' &rarr; ' || gbqs_h(tp) || '</div>'
      || '<div style="color:#5c6b85;font-size:12px;">' || gbqs_h(hire_cert_point_label(c->>'point_code', c->>'point_text'))
      || coalesce(' &middot; ' || to_char((c->>'event_at')::timestamptz at time zone 'UTC','DD-MM-YYYY HH24:MI') || ' UTC','') || '</div>'
      || '<div style="font-size:13px;margin-top:4px;">ROB: ' || gr || '</div></td></tr>';
  end loop;

  if iss->>'kind' = 'surveyor' then
    select * into co from surveyor_companies where id = (iss->>'company_id')::uuid;
    v_logo := sa_cert_logo_url((sv->>'id')::uuid);
    select coalesce(nullif((select value from system_settings where key = 'sa_report_view_base'), ''), (select value from system_settings where key = 'sa_report_link_base')) || '?t=' || l.token
      into v_view from sa_report_links l where l.survey_id = (sv->>'id')::uuid and l.revoked_at is null order by l.created_at desc limit 1;
    select array_agg(distinct lower(trim(e))) into v_cc from jsonb_array_elements_text(coalesce(iss->'client_emails','[]')) e where trim(e) like '%@%';
    v_to := v_cc[1]; v_cc := v_cc[2:];
    if nullif(trim(co.office_email),'') is not null and not (lower(trim(co.office_email)) = any(coalesce(v_cc,'{}') || coalesce(v_to,''))) then
      v_cc := coalesce(v_cc,'{}') || lower(trim(co.office_email));
    end if;
    if v_to is null then v_to := nullif(lower(trim(co.office_email)),''); v_cc := array_remove(v_cc, v_to); end if;
    body := '<p style="margin:0 0 6px;font-size:18px;font-weight:bold;">Your bunker certificate' || case when n > 1 then 's are' else ' is' end || ' ready</p>'
      || '<p style="margin:0 0 18px;">Dear ' || gbqs_h(coalesce(nullif(iss->>'client_contact',''), 'Sir / Madam')) || ',<br>Based on the ' || gbqs_h(lower(hire_cert_survey_label(sv->>'survey_type'))) || ' of <strong>' || gbqs_h(sv->>'vessel_name')
      || '</strong> (' || gbqs_h(sv->>'gbqs_reference') || ') and the figures declared by the Master, we have issued the following certificate' || case when n > 1 then 's' else '' end
      || '. The PDF' || case when n > 1 then 's are' else ' is' end || ' attached to this e-mail.</p>'
      || '<table width="100%" cellpadding="0" cellspacing="0" style="background:#f5f8fc;border:1px solid #e2e7f2;border-radius:8px;margin:0 0 20px;font-size:14px;">' || rows || '</table>'
      || case when v_view is not null then '<p style="margin:0 0 14px;">' || gbqs_mail_btn(v_view, 'View the survey report', 'secondary') || '</p>' else '' end
      || '<table width="100%" cellpadding="0" cellspacing="0" style="background:#e9f5fb;border-left:4px solid #2a8fc4;margin:6px 0 18px;"><tr><td style="padding:14px 16px;font-size:13px;line-height:20px;">'
      || 'Each certificate carries a verification code. To verify a certificate, simply reply to this e-mail quoting the certificate number.</td></tr></table>'
      || '<p style="margin:0;">Best regards,<br><strong>' || gbqs_h(co.name) || '</strong></p>';
    v_ship := nullif(lower(trim(d->'case'->>'request_to')),'');
    v_kindword := case when (sv->>'is_on_hire')::boolean then 'on-hire' else 'off-hire' end;
    if v_ship is not null and v_ship <> coalesce(v_to,'') and not (v_ship = any(coalesce(v_cc,'{}'))) then

create or replace function public.sa_cert_can(p_case uuid)
returns boolean language sql stable security definer set search_path to 'public' as $function$
  select exists (select 1 from hire_cert_cases cc join surveyor_jobs j on j.id = cc.surveyor_job_id
                 where cc.id = p_case and cc.issuer = 'surveyor' and j.company_id = sa_my_company_id()
                   and sa_job_visible(j.company_id, j.assigned_user_id, j.location_id)) $function$;

create or replace function public.sa_cert_job_can(p_job uuid)
returns boolean language sql stable security definer set search_path to 'public' as $function$
  select exists (select 1 from surveyor_jobs j where j.id = p_job and j.company_id = sa_my_company_id()
                   and sa_job_visible(j.company_id, j.assigned_user_id, j.location_id)) $function$;

create or replace function public.sa_try_uuid(p text)
returns uuid language plpgsql immutable as $function$
begin return p::uuid; exception when others then return null; end $function$;

-- 9. Engelse teksten voor de controle
create or replace function public.sa_cert_check_en(p_check jsonb)
returns jsonb language sql immutable set search_path to 'public' as $function$
  select coalesce(p_check, '{}'::jsonb) || jsonb_build_object(
    'direction_text', case p_check->>'direction' when 'survey_before' then 'Survey before the handover: consumption is deducted'
      when 'survey_after' then 'Survey after the handover: consumption is added back' when 'same' then 'Handover at the time of the survey: no correction'
      else 'Handover time not entered yet' end,
    'warnings', coalesce((select jsonb_agg(w || jsonb_build_object('text_en',
       case w->>'code'
         when 'no_time' then 'Handover date and time (UTC) is missing'
         when 'parties' then case when w->>'text' like '%gelijk%' then 'From and to party are the same' else 'From or to party is missing' end
         when 'interval' then 'Interval ' || coalesce(p_check->>'interval_text','') || ' is longer than allowed – please check'
         when 'retro' then 'Retrospective handover: ' || coalesce(p_check->>'interval_text','') || ' before the survey (stated on the certificate)'
         when 'evidence' then 'No reply from the Master recorded – enter the reply time or attach the e-mail'
         when 'survey_moved' then 'Survey time differs more than 24 hours from the signatures in the report – check the year'
         when 'cons_missing' then (w->>'grade') || ': consumption not entered (0 is allowed)'
         when 'negative' then (w->>'grade') || ': calculated ROB is negative'
         when 'dev' then (w->>'grade') || ': average ' || coalesce(gg->>'avg_day','?') || ' MT/day is '
              || case when (gg->>'dev_pct')::numeric > 0 then '+' else '' end || coalesce(gg->>'dev_pct','?') || '% against '
              || case gg->>'ref_label' when 'gem. eerdere certificaten' then 'the average of your earlier certificates' when 'CP-verbruik' then 'the CP consumption' else coalesce(gg->>'ref_label','the reference') end
              || ' (' || coalesce(gg->>'ref','?') || ')'
         when 'avg_high' then (w->>'grade') || ': average consumption above 150 MT/day – check the unit'
         else w->>'text' end))
       from jsonb_array_elements(coalesce(p_check->'warnings','[]')) w
       left join lateral (select x from jsonb_array_elements(coalesce(p_check->'grades','[]')) x where x->>'grade' = w->>'grade' limit 1) gx(gg) on true), '[]'::jsonb)) $function$;

-- 10. Portaalfuncties
create or replace function public.sa_portal_cert_get(p_job uuid)
returns jsonb language plpgsql stable security definer set search_path to 'public' as $function$
declare j surveyor_jobs; s surveys; cid uuid; d jsonb; certs jsonb := '[]'; c jsonb; v_issued timestamptz;
begin
  if not sa_cert_job_can(p_job) then raise exception 'Job not found or no access'; end if;
  select * into j from surveyor_jobs where id = p_job;
  select * into s from surveys where id = j.survey_id;
  select id into cid from hire_cert_cases where surveyor_job_id = p_job;
  v_issued := coalesce(s.issued_at, j.updated_at);
  if cid is null then
    return jsonb_build_object('case', null, 'job', jsonb_build_object('id', j.id, 'reference', j.reference, 'survey_type', j.survey_type, 'status', j.status, 'vessel', j.vessel_name),
      'eligible', j.survey_type in ('on_hire_bunker','off_hire_bunker','on_hire_bc','off_hire_bc'),
      'can_create', j.survey_type in ('on_hire_bunker','off_hire_bunker','on_hire_bc','off_hire_bc') and s.status in ('issued','invoiced','closed')
                    and v_issued > now() - interval '30 days');
  end if;
  d := hire_cert_case_get(cid);
  for c in select * from jsonb_array_elements(d->'certificates') loop
    certs := certs || (c || jsonb_build_object('check', sa_cert_check_en(c->'check')));
  end loop;
  return (d - 'certificates') || jsonb_build_object('certificates', certs,
    'job', jsonb_build_object('id', j.id, 'reference', j.reference, 'survey_type', j.survey_type, 'status', j.status, 'vessel', j.vessel_name,
       'port', (select naam from havens where id = j.haven_id), 'client', (select name from surveyor_clients where id = j.surveyor_client_id)),
    'evidence_name', (select evidence_name from hire_cert_cases where id = cid),
    'can_edit', (d->'case'->>'status') not in ('issued','cancelled'));
end $function$;

create or replace function public.sa_portal_cert_create(p_job uuid)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare g jsonb;
begin
  g := sa_portal_cert_get(p_job);
  if g->'case' is not null and g->'case' <> 'null'::jsonb then return g; end if;
  if not coalesce((g->>'can_create')::boolean, false) then raise exception 'A certificate can only be created for an issued on-/off-hire job of the last 30 days'; end if;
  perform hire_cert_ensure_case((select survey_id from surveyor_jobs where id = p_job));
  return sa_portal_cert_get(p_job);
end $function$;

create or replace function public.sa_portal_cert_request(p_case uuid, p_to text, p_cc text default null, p_reminder boolean default false)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare st text; v_by text;
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  select status into st from hire_cert_cases where id = p_case;
  if st in ('issued','cancelled') then raise exception 'This certificate case is closed'; end if;
  if coalesce(trim(p_to),'') !~ '^[^\s@,;]+@[^\s@,;]+\.[^\s@,;]+$' then raise exception 'Please enter a valid e-mail address of the vessel'; end if;
  select coalesce(nullif(full_name,''), email) into v_by from surveyor_company_users where user_id = auth.uid();
  perform hire_cert_request(p_case, p_to, p_cc, v_by, p_reminder);
  return jsonb_build_object('ok', true);
end $function$;

create or replace function public.sa_portal_cert_save(p jsonb)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare cid uuid := sa_try_uuid(p->>'case_id'); st text; pp jsonb; jid uuid;
begin
  if cid is null or not sa_cert_can(cid) then raise exception 'Certificate not found or no access'; end if;
  select status, surveyor_job_id into st, jid from hire_cert_cases where id = cid;
  if st = 'issued' then raise exception 'The certificates have been issued and can no longer be changed'; end if;
  if st = 'cancelled' then raise exception 'This certificate case is closed'; end if;
  for pp in select * from jsonb_array_elements(coalesce(p->'parties','[]')) loop
    if coalesce(trim(pp->>'company'),'') = '' then raise exception 'Every party needs a company name'; end if;
    if coalesce(pp->>'key','') = '' then raise exception 'Party without key'; end if;
  end loop;
  perform hire_cert_save(p - 'survey_completed_at');
  return sa_portal_cert_get(jid);
end $function$;

create or replace function public.sa_portal_cert_evidence(p_case uuid, p_path text, p_name text)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare jid uuid;
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  if p_path is not null and split_part(p_path, '/', 1) <> p_case::text then raise exception 'Invalid file'; end if;
  if p_path is not null and not exists (select 1 from storage.objects where bucket_id = 'sa-cert-evidence' and name = p_path) then raise exception 'File not found'; end if;
  update hire_cert_cases set evidence_path = case when p_path is null then null else 'sa-cert-evidence/' || p_path end,
         evidence_name = case when p_path is null then null else left(coalesce(p_name, 'master_reply'), 200) end,
         master_reply_at = coalesce(master_reply_at, case when p_path is not null then now() end),
         status = case when status in ('open','requested') and p_path is not null then 'received' else status end, updated_at = now()
   where id = p_case returning surveyor_job_id into jid;
  return sa_portal_cert_get(jid);
end $function$;

create or replace function public.sa_portal_cert_issue(p_case uuid)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare v_by text; r jsonb; e text; jid uuid;
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  if exists (select 1 from hire_cert_cases where id = p_case and status = 'issued') then raise exception 'Already issued'; end if;
  if not exists (select 1 from hire_certificates where case_id = p_case and status = 'draft') then raise exception 'There is no certificate to issue'; end if;
  select string_agg(hc.letter || ': ' || (w->>'text_en'), '; ') into e
    from hire_certificates hc, jsonb_array_elements(sa_cert_check_en(hire_cert_calc(hc.id))->'warnings') w
   where hc.case_id = p_case and hc.status = 'draft' and w->>'level' = 'error';
  if e is not null then raise exception 'Please correct first – %', e; end if;
  select coalesce(nullif(full_name,''), email) into v_by from surveyor_company_users where user_id = auth.uid();
  r := hire_cert_issue(p_case, v_by);
  select surveyor_job_id into jid from hire_cert_cases where id = p_case;
  return sa_portal_cert_get(jid) || jsonb_build_object('issued', r->'certificates');
end $function$;

create or replace function public.sa_portal_cert_resend(p_case uuid)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  if not exists (select 1 from hire_cert_cases where id = p_case and status = 'issued') then raise exception 'Not issued yet'; end if;
  return hire_cert_resend(p_case);
end $function$;

create or replace function public.sa_portal_cert_not_needed(p_case uuid)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare jid uuid;
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  update hire_cert_cases set status = 'cancelled', updated_at = now() where id = p_case and status in ('open','requested','received')
  returning surveyor_job_id into jid;
  if jid is null then raise exception 'Only an open case can be closed'; end if;
  return sa_portal_cert_get(jid);
end $function$;

create or replace function public.sa_portal_cert_reopen(p_case uuid)
returns jsonb language plpgsql security definer set search_path to 'public' as $function$
declare jid uuid;
begin
  if not sa_cert_can(p_case) then raise exception 'Certificate not found or no access'; end if;
  update hire_cert_cases set status = case when master_reply_at is not null then 'received' when requested_at is not null then 'requested' else 'open' end, updated_at = now()
   where id = p_case and status = 'cancelled' returning surveyor_job_id into jid;
  if jid is null then raise exception 'Only a closed case can be reopened'; end if;
  return sa_portal_cert_get(jid);
end $function$;

create or replace view public.portal_hire_certificates as
 SELECT hc.id, cc.order_id, cc.survey_id, s.gbqs_reference, hc.letter, hc.cert_number, hc.kind,
    CASE WHEN hc.kind = 'delivery' THEN 'On-hire Bunker Certificate' ELSE 'Off-hire Bunker Certificate' END AS title,
    hire_cert_role_label(pf.role, pf.role_label) AS from_role, pf.company AS from_company,
    hire_cert_role_label(pt.role, pt.role_label) AS to_role, pt.company AS to_company,
    hc.cp_date, hire_cert_point_label(hc.point_code, hc.point_text) AS point, hc.event_at, hc.no_correction, hc.issued_at, hc.verify_code,
    (hc.pdf_path IS NOT NULL) AS has_pdf,
    (SELECT string_agg(g.grade || ' ' || to_char(g.calc_mt, 'FM999G990D000') || ' MT', ' · ' ORDER BY g.sort) FROM hire_certificate_grades g WHERE g.certificate_id = hc.id) AS rob_text,
    o.client_id
   FROM hire_certificates hc
     JOIN hire_cert_cases cc ON cc.id = hc.case_id
     JOIN surveys s ON s.id = cc.survey_id
     JOIN orders o ON o.id = cc.order_id
     LEFT JOIN hire_cert_parties pf ON pf.id = hc.from_party_id
     LEFT JOIN hire_cert_parties pt ON pt.id = hc.to_party_id
  WHERE hc.status = 'issued' AND o.client_id IN (SELECT portal_my_client_ids())
UNION ALL
 SELECT hc.id, NULL::uuid AS order_id, x.id AS survey_id,
    COALESCE(NULLIF(btrim(x.report_reference), ''), s.gbqs_reference) AS gbqs_reference, hc.letter, hc.cert_number, hc.kind,
    CASE WHEN hc.kind = 'delivery' THEN 'On-hire Bunker Certificate' ELSE 'Off-hire Bunker Certificate' END AS title,
    hire_cert_role_label(pf.role, pf.role_label), pf.company, hire_cert_role_label(pt.role, pt.role_label), pt.company,
    hc.cp_date, hire_cert_point_label(hc.point_code, hc.point_text), hc.event_at, hc.no_correction, hc.issued_at, hc.verify_code,
    (hc.pdf_path IS NOT NULL),
    (SELECT string_agg(g.grade || ' ' || to_char(g.calc_mt, 'FM999G990D000') || ' MT', ' · ' ORDER BY g.sort) FROM hire_certificate_grades g WHERE g.certificate_id = hc.id),
    x.client_id
   FROM hire_certificates hc
     JOIN hire_cert_cases cc ON cc.id = hc.case_id AND cc.issuer = 'surveyor'
     JOIN surveys s ON s.id = cc.survey_id
     JOIN external_surveys x ON x.surveyor_job_id = cc.surveyor_job_id
     LEFT JOIN hire_cert_parties pf ON pf.id = hc.from_party_id
     LEFT JOIN hire_cert_parties pt ON pt.id = hc.to_party_id
  WHERE hc.status = 'issued' AND x.client_id IN (SELECT portal_my_client_ids());

  n := replace(d, '''share'', j.parties->>''share'',',
    '''share'', j.parties->>''share'',
      ''cert'', (select jsonb_build_object(''case_id'', hcx.id, ''status'', hcx.status,
         ''numbers'', (select string_agg(hc2.cert_number, '', '' order by hc2.letter) from hire_certificates hc2 where hc2.case_id = hcx.id and hc2.status = ''issued''))
       from hire_cert_cases hcx where hcx.surveyor_job_id = j.id),');
  if n = d then raise exception 'anker sa_portal_reports niet gevonden'; end if;
  execute n;
end $$;
